        function wineApp() {
            return {
                lang: localStorage.getItem('grandcru_lang') || 'th',
                isLoggedIn: false,
                userName: '',
                userEmail: '',
                userRole: 'customer',
                currentTab: 'shop',
                wines: [],
                brandList: [],
                filters: { price: '', type: '', vintage: '', country: '', brand: '' },
                interestList: [],
                interestCount: 0,
                showAuthModal: false,
                authMode: 'login',
                loginForm: { email: '', password: '' },
                signupForm: { first_name: '', last_name: '', email: '', password: '' },
                selectedWine: null,
                interestQuantity: 1,
                reportPeriod: 'monthly',
                reports: [],

                // Admin Wine Form States
                showWineFormModal: false,
                isEditingWine: false,
                wineForm: {
                    id: null, name: '', price: '', vintage: '', city: '', country: '',
                    type: 'Red Wine', variety: '', alcohol: '', size: '', brand: '',
                    color: '', sweetness: '', taste: '', aroma: '', food_pairing: '',
                    description: '', image_url: ''
                },

                init() {
                    this.fetchWines();
                    this.checkSession();
                },

                setLang(newLang) {
                    this.lang = newLang;
                    localStorage.setItem('grandcru_lang', newLang);
                },

                t(key) {
                    return translations[this.lang][key] || key;
                },

                translateStatus(status) {
                    const map = {
                        'กำลังจัดสั่งซื้อ': this.t('statusOrdering'),
                        'ระหว่างขนส่ง': this.t('statusShipping'),
                        'กำลังจัดจำหน่าย': this.t('statusSelling'),
                        'จำหน่ายแล้ว': this.t('statusSold')
                    };
                    return map[status] || status;
                },

                async checkSession() {
                    try {
                        const res = await fetch('/api/interests/list');
                        const data = await res.json();
                        if (Array.isArray(data)) {
                            this.isLoggedIn = true;
                            this.interestList = data;
                            this.interestCount = data.length;
                        }
                    } catch(e) {}
                },

                async fetchWines() {
                    let params = new URLSearchParams();
                    for (let key in this.filters) {
                        if (this.filters[key]) params.append(key, this.filters[key]);
                    }
                    const res = await fetch('/api/wines?' + params.toString());
                    const data = await res.json();
                    this.wines = data;
                    
                    const brands = [...new Set(data.map(w => w.brand))].filter(Boolean);
                    this.brandList = brands;
                },

                async login() {
                    const res = await fetch('/api/login', {
                        method: 'POST',
                        headers: { 'Content-Type': 'application/json' },
                        body: JSON.stringify(this.loginForm)
                    });
                    const data = await res.json();
                    if (data.success) {
                        this.isLoggedIn = true;
                        this.userName = data.name;
                        this.userRole = data.role;
                        this.showAuthModal = false;
                        this.fetchInterests();
                        this.fetchReports();
                        alert(this.t('msgLoginSuccess'));
                    } else {
                        alert(data.message);
                    }
                },

                async signup() {
                    const res = await fetch('/api/signup', {
                        method: 'POST',
                        headers: { 'Content-Type': 'application/json' },
                        body: JSON.stringify(this.signupForm)
                    });
                    const data = await res.json();
                    if (data.success) {
                        alert(data.message);
                        this.authMode = 'login';
                    } else {
                        alert(data.message);
                    }
                },

                async socialLogin(provider) {
                    const res = await fetch('/api/social-login', {
                        method: 'POST',
                        headers: { 'Content-Type': 'application/json' },
                        body: JSON.stringify({ provider })
                    });
                    const data = await res.json();
                    if (data.success) {
                        this.isLoggedIn = true;
                        this.userName = data.name;
                        this.userRole = data.role;
                        this.showAuthModal = false;
                        this.fetchInterests();
                        this.fetchReports();
                        alert(this.t('msgLoginSuccess') + ' (' + provider + ')');
                    }
                },

                async logout() {
                    await fetch('/api/logout');
                    this.isLoggedIn = false;
                    this.userName = '';
                    this.userEmail = '';
                    this.userRole = 'customer';
                    this.interestList = [];
                    this.interestCount = 0;
                    this.currentTab = 'shop';
                    alert(this.t('msgLogoutSuccess'));
                },

                async fetchInterests() {
                    const res = await fetch('/api/interests/list');
                    const data = await res.json();
                    if (Array.isArray(data)) {
                        this.interestList = data;
                        this.interestCount = data.length;
                    }
                },

                async saveInterest() {
                    if (!this.isLoggedIn) {
                        alert(this.t('msgLoginFirst'));
                        this.showAuthModal = true;
                        return;
                    }
                    const res = await fetch('/api/interest', {
                        method: 'POST',
                        headers: { 'Content-Type': 'application/json' },
                        body: JSON.stringify({ wine_id: this.selectedWine.id, quantity: this.interestQuantity })
                    });
                    const data = await res.json();
                    if (data.success) {
                        alert(data.message);
                        this.selectedWine = null;
                        this.fetchInterests();
                    } else {
                        alert(data.message);
                    }
                },

                async updateOrderStatus(interestId, status) {
                    const res = await fetch('/api/admin/update-status', {
                        method: 'POST',
                        headers: { 'Content-Type': 'application/json' },
                        body: JSON.stringify({ interest_id: interestId, status: status })
                    });
                    const data = await res.json();
                    if (data.success) {
                        alert(this.t('msgStatusUpdated'));
                        this.fetchInterests();
                    }
                },

                async fetchReports() {
                    const res = await fetch(`/api/admin/reports?period=${this.reportPeriod}`);
                    const data = await res.json();
                    if (Array.isArray(data)) {
                        this.reports = data;
                    }
                },

                openWineForm() {
                    this.isEditingWine = false;
                    this.wineForm = {
                        id: null, name: '', price: '', vintage: '', city: '', country: '',
                        type: 'Red Wine', variety: '', alcohol: '', size: '', brand: '',
                        color: '', sweetness: '', taste: '', aroma: '', food_pairing: '',
                        description: '', image_url: ''
                    };
                    this.showWineFormModal = true;
                },

                editWine(wine) {
                    this.isEditingWine = true;
                    this.wineForm = { ...wine };
                    this.showWineFormModal = true;
                },

                async saveWineForm() {
                    const method = this.isEditingWine ? 'PUT' : 'POST';
                    const res = await fetch('/api/admin/wine', {
                        method: method,
                        headers: { 'Content-Type': 'application/json' },
                        body: JSON.stringify(this.wineForm)
                    });
                    const data = await res.json();
                    if (data.success) {
                        alert(data.message);
                        this.showWineFormModal = false;
                        this.fetchWines();
                    } else {
                        alert(this.t('msgError') + data.message);
                    }
                },

                async deleteWine(wineId) {
                    if (!confirm(this.t('msgConfirmDelete'))) return;
                    const res = await fetch(`/api/admin/wine?id=${wineId}`, { method: 'DELETE' });
                    const data = await res.json();
                    if (data.success) {
                        this.fetchWines();
                    } else {
                        alert(this.t('msgDeleteError') + data.message);
                    }
                },

                openWineDetail(wine) {
                    this.selectedWine = wine;
                    this.interestQuantity = 1;
                },

                getPageTitle() {
                    const titles = {
                        'shop': this.t('pageTitleShop'),
                        'profile': this.t('pageTitleProfile'),
                        'interests': this.t('pageTitleInterests'),
                        'admin': this.t('pageTitleAdmin')
                    };
                    return titles[this.currentTab] || '';
                },

                getCleanImageUrl(url) {
                    if (!url) return 'https://images.unsplash.com/photo-1510812431401-41d2bd2722f3?w=500&auto=format&fit=crop&q=60';
                    if (url.includes('drive.google.com')) {
                        const match = url.match(/\/d\/([a-zA-Z0-9_-]+)/);
                        if (match && match[1]) {
                            return `https://lh3.googleusercontent.com/d/${match[1]}`;
                        }
                    }
                    return url;
                },

                formatAlcohol(alc) {
                    if (!alc) return '-';
                    if (typeof alc === 'number') {
                        return (alc <= 1 ? (alc * 100).toFixed(1) : alc.toFixed(1)) + '%';
                    }
                    return alc;
                }
            }
        }
